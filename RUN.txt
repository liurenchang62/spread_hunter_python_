# 标的：TRX 永续合约（各所最小下单量 < 1 USDT，手续费极低）
#
# 分组说明：
#   [安全组] 只读/公开接口，不涉及任何资金操作，始终自动执行
#   [资金组] 会产生真实委托或成交，需要加 --money 参数并逐步确认
#   [划转组] 涉及账户间资金转移，需要加 --transfer 参数

# ── 套件方式运行

# 只跑安全组（API 连通性、账户查询、订单薄、资金费率）
python -m test_live.run_all

# 包含资金组（安全组 + 限价单撤单 + 最小市价单）
python -m test_live.run_all --money

# 包含划转组（在 --money 基础上再加划转测试，干跑模式）
python -m test_live.run_all --money --transfer

# 只测某一所
python -m test_live.run_all --ex gate
python -m test_live.run_all --ex binance --money

# ── 单项实盘测试（安全组，无资金风险）

# [1] API 连通性：公开接口延迟 + API Key 签名验证 + TRX/BTC 价格查询
python -m test_live.test_connectivity

# [2] 账户信息：合约余额、现货余额、持仓、挂单（只读）
python -m test_live.test_account
python -m test_live.test_account --ex okx

# [3] 订单薄：TRX 20档深度、买卖价差、滑点估算（公开接口）
python -m test_live.test_orderbook
python -m test_live.test_orderbook --ex gate

# [4] 资金费率 & 合约规格：taker 费率、资金费率、60s 持仓成本估算
python -m test_live.test_funding
python -m test_live.test_funding --ex bitget

# ── 单项实盘测试（资金组，⚠️ 涉及真实资金）

# [5] 限价单撤单：以市价×50% 挂限价买单（不会成交），确认后撤单
#     风险：临时占用少量保证金（几秒），无成交风险
python -m test_live.test_cancel
python -m test_live.test_cancel --ex binance

# [6] 最小市价单：买入最小量 TRX 并立即平仓
#     费用：约 0.05–0.15 USDT/所（2×taker 手续费）
python -m test_live.test_order
python -m test_live.test_order --ex okx

# [7] 合约→现货划转：默认干跑，--execute 才实际转账
python -m test_live.test_transfer                           # 干跑（显示计划，不执行）
python -m test_live.test_transfer --execute                 # 实际划转 10 USDT/所（需逐所确认）
python -m test_live.test_transfer --ex gate --execute --amount 10

# 单独运行完整预检（与 main.py --live 启动前检查一致，便于单独排查）
python -m test_live.preflight

#  资金再平衡（rebalance/ 目录）
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
# 触发条件（所有参数均为比例，见 trader/config.py）：
#   REBALANCE_FLOOR_PCT=20%  — 某所现金 < 总现金×20% → 触发再平衡
#   CASH_RATIO_MIN=30%       — 整体现金/权益 < 30% → 进入仅监控模式
#   CASH_RATIO_RESUME=40%    — 整体现金/权益 > 40% → 退出仅监控模式（迟滞）
# 实盘主程序（--live）自动每 4h 后台检查一次（RebalanceSupervisor）
# 手动工具始终使用实盘 API Key（测试网不支持提现 API）

# [1] 查看各交易所余额 + 再平衡/流动性状态（只读）
python -m rebalance.check_balances

# [2] 查看各交易所提现手续费 + 路径对比（只读）
python -m rebalance.check_fees                          # 查询全部 4 所
python -m rebalance.check_fees --ex gate                # 只查 gate

# [3] 干跑：显示余额、触发判断、路径规划、提现计划（不执行任何操作）
python -m rebalance.run

# [4] 实际执行：输入 YES 全局确认 + 每笔提现逐一确认
#     注意：执行前确保主程序未在活跃开仓，或在低活跃时段运行
python -m rebalance.run --execute
